```python
class AIPerformanceTracker:
    def __init__(self):
        self.claude_record = {"wins": 0, "losses": 0, "total": 0}
        self.gpt4_record = {"wins": 0, "losses": 0, "total": 0}
        self.gemini_record = {"wins": 0, "losses": 0, "total": 0}

    def record_trade_outcome(self, trade, outcome):
        """
        Record which AIs voted YES and track their accuracy
        """
        if trade.claude_vote == "EXECUTE":
            self.claude_record["total"] += 1
            if outcome == "WIN":
                self.claude_record["wins"] += 1
            else:
                self.claude_record["losses"] += 1

        # Same for GPT-4 and Gemini

    def get_ai_accuracy(self, ai_name):
        """
        Calculate win rate for each AI
        """
        record = getattr(self, f"{ai_name}_record")
        if record["total"] == 0:
            return 0
        return record["wins"] / record["total"]

    def get_best_ai(self):
        """
        Find which AI has highest accuracy
        """
        claude_acc = self.get_ai_accuracy("claude")
        gpt4_acc = self.get_ai_accuracy("gpt4")
        gemini_acc = self.get_ai_accuracy("gemini")

        return max([
            ("Claude", claude_acc),
            ("GPT-4", gpt4_acc),
            ("Gemini", gemini_acc)
        ], key=lambda x: x[1])
```

```python
def calculate_final_score(base_score, ai_adjustments):
    """
    Combine base SCOB-Flip score with AI adjustments
    """
    # Base score from SCOB-Flip Integration
    flip_score = base_score["flip"]  # 0-50
    scob_score = base_score["scob"]  # 0-50
    base_total = flip_score + scob_score  # 0-100

    # AI adjustments
    claude_adjustment = ai_adjustments["claude"]  # -10 to +10
    gpt4_adjustment = ai_adjustments["gpt4"]      # -5 to +5
    gemini_adjustment = ai_adjustments["gemini"]  # -5 to +5

    total_adjustment = (
        claude_adjustment +
        gpt4_adjustment +
        gemini_adjustment
    )

    # Final score (capped at 0-100)
    final_score = max(0, min(100, base_total + total_adjustment))

    return {
        "base_score": base_total,
        "ai_adjustment": total_adjustment,
        "final_score": final_score,
        "grade": get_grade(final_score)
    }

def get_grade(score):
    if score >= 95:
        return "PERFECT+"
    elif score >= 90:
        return "PERFECT"
    elif score >= 80:
        return "EXCELLENT"
    elif score >= 70:
        return "STRONG"
    elif score >= 60:
        return "DECENT"
    else:
        return "WEAK"
```

### **Session Optimization Learning:**

**The system LEARNS which pairs perform best in which sessions:**

```python
class SessionOptimizer:
    def __init__(self):
        self.performance_data = {
            "asian": {},
            "london": {},
            "ny": {}
        }

    def record_trade(self, session, pair, outcome, r_multiple):
        """
        Track performance by session and pair
        """
        if pair not in self.performance_data[session]:
            self.performance_data[session][pair] = {
                "wins": 0,
                "losses": 0,
                "total_r": 0
            }

        data = self.performance_data[session][pair]
        if outcome == "WIN":
            data["wins"] += 1
            data["total_r"] += r_multiple
        else:
            data["losses"] += 1
            data["total_r"] -= 1  # Assume 1R loss

    def get_best_pairs_for_session(self, session):
        """
        Return top 3 pairs for given session
        """
        pairs = self.performance_data[session]
        ranked = []

        for pair, data in pairs.items():
            total_trades = data["wins"] + data["losses"]
            if total_trades >= 10:  # Minimum sample size
                win_rate = data["wins"] / total_trades
                avg_r = data["total_r"] / total_trades
                score = win_rate * avg_r  # Combined metric
                ranked.append((pair, score, win_rate, avg_r))

        # Sort by score
        ranked.sort(key=lambda x: x[1], reverse=True)

        return ranked[:3]  # Top 3 pairs

    def get_recommendation(self, session, pair):
        """
        Should we trade this pair in this session?
        """
        if pair not in self.performance_data[session]:
            return "NEUTRAL"  # No data yet

        data = self.performance_data[session][pair]
        total = data["wins"] + data["losses"]

        if total < 10:
            return "NEUTRAL"  # Not enough data

        win_rate = data["wins"] / total

        if win_rate >= 0.80:
            return "HIGHLY RECOMMENDED"
        elif win_rate >= 0.70:
            return "RECOMMENDED"
        elif win_rate >= 0.60:
            return "ACCEPTABLE"
        else:
            return "AVOID"
```
